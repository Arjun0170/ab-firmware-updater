cmake_minimum_required(VERSION 3.16)
project(ab_firmware_updater C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(updater src/updater.c)
target_include_directories(updater PUBLIC include)
target_compile_options(updater PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(flash_model sim/flash_model.c)
target_include_directories(flash_model PUBLIC sim)
target_link_libraries(flash_model PUBLIC updater)
target_compile_options(flash_model PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(updater_sim sim/main.c)
target_link_libraries(updater_sim PRIVATE flash_model)
add_executable(updater_tests tests/test_main.c)
target_link_libraries(updater_tests PRIVATE flash_model)
enable_testing()
add_test(NAME updater_tests COMMAND updater_tests)
